    Q["用户查询 query"]:::in

    subgraph S1["阶段 1 · 查询理解  understand_query"]
        direction TB
        U1["归一化 + 意图识别
definition / constraint / test_method / comparison / procedure / evidence"]:::p1
        U2["目标节点链接
查询词 → ontology 节点别名（正/反向匹配 + 置信度）"]:::onto
        U3["别名展开 · 答案契约 · must/should 词

        U4["QueryFrame
intent · target_objects · retrieval_channels · answer_strategy"]:::frame
        U1 --> U2 --> U3 --> U4
    end

    subgraph S2["阶段 2 · 多通道检索  hybrid_retrieve"]
        direction TB
        B["分支 A · 内存基线 retrieve()
object_card · fact · table · evidence · keyword · semantic"]:::p2
        L["分支 B1 · lexical 词面
FTS trigram 全文检索 · w = 1.00"]:::p2
        V["分支 B2 · vector 语义
远程嵌入 bge-small-zh-v1.5 余弦相似度 · w = 0.95"]:::emb
        G["分支 B3 · graph 走图
ontology 关系 BFS 遍历（max_depth = 2）· w = 0.85"]:::onto
        M["跨索引合并 + 交叉印证
multi_adapter_corroboration 加分"]:::onto
        R["确定性重排  DeterministicReranker
目标节点 / 事实类型 / 证据绑定 / 答案形态"]:::p2
        B --> M
        L --> M
        V --> M
        G --> M
        M --> R
    end

    subgraph S3["阶段 3 · Context Pack 组装  build_context_pack"]
        direction TB
        C1["选取 目标对象 / 关系 / 节点卡 / 事实 / 证据"]:::p3
        C2["溯源链  evidence → facts → cards → nodes"]:::onto
        C3["hidden_context · warnings · knowledge_gaps"]:::p3
        C1 --> C2 --> C3
    end

    subgraph S4["阶段 4 · 证据充分性判定  judge_context_pack"]
        direction TB
        J["加权打分
0.45 × 形态覆盖 + 0.30 × 证据 + 0.15 × 绑定 + 0.10 × 节点"]:::p4
        JD{"充分性判定"}:::dec
        J --> JD
    end

    subgraph S5["阶段 5 · 答案生成  answer_query"]
        direction TB
        AS1["answer_with_evidence
直接回答 + 标注出处"]:::p5
        AS2["answer_with_caution
谨慎回答 + 标注不确定与限制"]:::p5
        AS3["abstain
不编造 / 通用知识补充（明确分隔）"]:::p5
    end
